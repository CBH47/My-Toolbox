class MyServerCallbacks : public BLEServerCallbacks {
  void onConnect(BLEServer* pServer) { deviceConnected = true; Serial.println("Connected"); }
  void onDisconnect(BLEServer* pServer) {
    deviceConnected = false;
    Serial.println("Disconnected");
    pServer->getAdvertising()->start();
  }
};

class MyCallbacks : public BLECharacteristicCallbacks {
  void onWrite(BLECharacteristic *pChar) {
    String value = pChar->getValue().c_str();
    Serial.print("Received: ");
    Serial.println(value);
  }
};

void drawSafeText(String text, int x, int y) {
  int16_t x1, y1;
  uint16_t w, h;
  canvas.setTextSize(1); // match whatever size you're using
  canvas.getTextBounds(text, x, y, &x1, &y1, &w, &h);

  int endX = x1 + w;

  // If text starts before the seam but ends after it, it's straddling - push to next screen
  if (x1 < SEAM && endX > SEAM) {
    x = SEAM + 4; // small margin so it doesn't hug the bezel edge
  }
};

void pushCanvasToDisplays() {
  displayA.clearDisplay();
  displayB.clearDisplay();

  for (int y = 0; y < 64; y++) {
    for (int x = 0; x < 256; x++) {
      if (canvas.getPixel(x, y)) {
        if (x < SEAM) {
          displayA.drawPixel(x, y, SSD1306_WHITE);
        } else {
          displayB.drawPixel(x - SEAM, y, SSD1306_WHITE);
        }
      }
    }
  }

  displayA.display();
  displayB.display();
}

void btSetup(){
  BLEDevice::init("Cam's Toolbox");
  BLEServer *pServer = BLEDevice::createServer();
  pServer->setCallbacks(new MyServerCallbacks());

  BLEService *pService = pServer->createService(SERVICE_UUID);

  pCharacteristic = pService->createCharacteristic(
    CHARACTERISTIC_UUID,
    BLECharacteristic::PROPERTY_READ |
    BLECharacteristic::PROPERTY_WRITE |
    BLECharacteristic::PROPERTY_NOTIFY
  );
  pCharacteristic->setCallbacks(new MyCallbacks());
  pCharacteristic->addDescriptor(new BLE2902());
  pCharacteristic->setValue("Ready");

  pService->start();
  pServer->getAdvertising()->start();
  Serial.println("BLE advertising started");
}

 Serial.println("OLED found");
